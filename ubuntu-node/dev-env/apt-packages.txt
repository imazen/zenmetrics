# Division of labour, same as ~/work/zen/scripts/.mise.toml states for ARM:
#   mise owns user-space toolchains and CLIs   -> see mise.toml (arch-neutral)

#
# Derived from a real diff: `apt-mark showmanual` on the outgoing box (183) minus the new
# box (54) = 164 differing. That raw list is NOT what you want to install — see EXCLUDED.
#
#   sudo apt update && sudo apt install -y $(grep -vE '^\s*(#|$)' apt-packages.txt)

cmake_minimum_required(VERSION 3.16)
project(iqa_compete LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_TOOLS ON CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_DEVTOOLS OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_MANPAGES OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_SJPEG OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_SKCMS OFF CACHE BOOL "" FORCE)
set(JPEGXL_FORCE_SYSTEM_BROTLI ON CACHE BOOL "" FORCE)
set(JPEGXL_FORCE_SYSTEM_HWY ON CACHE BOOL "" FORCE)
set(JPEGXL_FORCE_SYSTEM_LCMS2 ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBJXL_SOURCE}" libjxl EXCLUDE_FROM_ALL)
add_executable(iqa-jxl libjxl_compute.cpp "${LIBJXL_SOURCE}/tools/ssimulacra2.cc" "${LIBJXL_SOURCE}/tools/no_memory_manager.cc")
target_include_directories(iqa-jxl PRIVATE "${LIBJXL_SOURCE}" "${LIBJXL_SOURCE}/lib/include" "${CMAKE_CURRENT_BINARY_DIR}/libjxl/lib/include")
target_link_libraries(iqa-jxl PRIVATE jxl_extras_nocodec-internal jxl_threads jxl_gauss_blur)

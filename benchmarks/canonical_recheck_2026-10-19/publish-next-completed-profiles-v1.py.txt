def run(*command):
    subprocess.run(command, check=True)

def load(path):
    return json.loads(path.read_text())

artifacts = []

def write(path, value):
    data = (json.dumps(value, indent=2, allow_nan=False)+'\n').encode()
    assert len(data) <= 30000
    with path.open('xb') as stream:
        stream.write(data)
    artifacts.append((path, 'provenance'))

def resource(path):
    receipts = [line for line in path.read_text().splitlines() if line.startswith('run-heavy: done ')]
    assert len(receipts) == 1 and 'rc=0 ' in receipts[0]
    return dict(path=str(path), sha256=sha(path), receipt=receipts[0])

for job in load(R/'nn-first-next-four-flags-v1/requests.json')['cases']:
    root = Path(job['profile'])
    receipt = load(root/'receipt.json')
    case = next(c for c in plan['cases'] if c['id'] == job['case'])
    binary = Path(case['native']['command'][0])
    assert sha(binary) == plan['bindings'][str(binary)] == debug['binary_sha256']
    assert receipt['case'] == case['id'] and receipt['threads'] == case['threads']
    assert receipt['plan_sha256'] == sha(plan_path) == job['plan_sha256']
    assert receipt['worker_sha256'] == sha(B/'profile_first_evaluation.py')
    assert receipt['profile_sha256'] == sha(root/'perf.data')
    assert receipt['report_sha256'] == sha(root/'report.txt')
    expected = [arg.format_map(dict(case, samples=1, output=str(root))) for arg in case['native']['command']]
    assert receipt['command'][-len(expected):] == expected
    report = (root/'report.txt').read_text()
    symbols = []
    for line in report.splitlines():
        found = re.match(r'^\s+([\d.]+)%\s+(\S+)\s+(\S+)\s+\[([.k])\]\s+(.*)$', line)
        if found:
            symbols.append(dict(self_percent=float(found[1]), process=found[2], object=found[3],
                                domain=found[4], symbol=found[5]))
    assert symbols
    lost = next(line for line in report.splitlines() if 'Total Lost Samples:' in line)
    log = root.with_suffix('.log')
    warnings = list(dict.fromkeys(line for line in log.read_text().splitlines() if 'could not read first record' in line))
    rss = re.search(r'Maximum resident set size \(kbytes\):\s*(\d+)', (root/'native-memory.txt').read_text())
    assert rss
    target = nn_output/(case['id']+'.json')
    write(target, dict(case=case['id'], raw_directory=str(root), receipt=receipt,
                       receipt_sha256=sha(root/'receipt.json'), native_binary_sha256=sha(binary),
                       native_max_process_rss_kbytes=int(rss[1]), top_self_symbols=symbols[:12],
                       lost_samples_header=lost, warnings=warnings, resource_receipt=resource(log),
                       shared_debug_audit=dict(file='../arniqa-first-profiles-2026-10-20/index.json', sha256=sha(debug_path)),
                       scope='Complete native probe includes immutable-model priming, first evaluation and score-stability controls. Partial source-line attribution. No isolated first-clock share, comparative timing or causal attribution.'))
    nn_index.append(dict(file=target.name, sha256=sha(target), case=case['id']))

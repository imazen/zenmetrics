def run(*command):
    subprocess.run(command, check=True)

def load(path):
    return json.loads(path.read_text())

def binding(path):
    return dict(path=str(path), sha256=sha(path))

for path_text, expected in summary['cell_results'].items():
    path = Path(path_text)
    assert sha(path) == expected
    value = load(path)
    assert value['fatal'] is False and set(value['arms']) == {'native', 'default'}
    for artifact, digest in value['artifacts'].items():
        assert sha(artifact) == digest, 'Retained HDR output changed: '+artifact
    case = specs[value['id']]
    if value['observation_complete']:
        assert value['absolute_difference'] == abs(value['arms']['native']['score']-value['arms']['default']['score'])

    for role in ['native', 'default']:
        original = value['arms'][role]
        memory = path.parent/role/'control-memory.txt'
        match = re.search(r'Maximum resident set size \(kbytes\):\s*(\d+)', memory.read_text())
        assert match
        arms[role] = {key:original[key] for key in ['returncode','complete','score','score_bits'] if key in original}
        arms[role].update(max_rss_kib=int(match[1]), memory=binding(memory), command=load(path.parent/role/'control-command.json'))

            raw = original['raw_row']

        elif role == 'default':
            arms[role]['samples'] = original.get('samples')

outputs = []

def write(path, value):
    data = (json.dumps(value, indent=2, allow_nan=False)+'\n').encode()
    assert len(data) <= 30000
    with path.open('xb') as stream:
        stream.write(data)
    outputs.append((path, 'provenance'))

shards = []
pending = []
def flush():
    if not pending:
        return
    path = folder/('cases-%02d.json'%len(shards))
    write(path, dict(cases=list(pending)))
    shards.append(dict(file=path.name, sha256=sha(path), cases=len(pending)))
    pending.clear()

for row in rows:
    if pending and len(json.dumps(dict(cases=pending+[row]), indent=2).encode()) > 24000:
        flush()
    pending.append(row)
flush()

                        retained_check_passes=sum(row['retained_check_passed'] is True for row in selected),
                        retained_check_misses=sum(row['retained_check_passed'] is False for row in selected),
                        max_absolute_difference=max(finite) if finite else None))

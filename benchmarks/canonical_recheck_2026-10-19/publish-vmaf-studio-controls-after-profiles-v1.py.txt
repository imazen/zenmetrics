import datetime

from pathlib import Path
import shutil

R = Path(__file__).parent
PUBLIC = Path('benchmarks/canonical_recheck_2026-10-19')
CONTROL = R/'vmaf-studio4k-current-score-controls-v1/result.json'

if len(sys.argv) == 1:
    while True:
        previous = (R/'publish-next-completed-profiles-v1.log').read_text()
        if 'NEXT_COMPLETED_PROFILES_VERIFIED' in previous and 'run-heavy: done rc=0 ' in previous and CONTROL.exists():
            break
        if 'run-heavy: done rc=' in previous and 'run-heavy: done rc=0 ' not in previous:
            raise SystemExit('Prior publication failed; preserve current-source controls')
        time.sleep(20)
    command = ['flock', str(Path.home()/'tmp/lilith-heavy.lock'), 'flock', str(Path.home()/'tmp/lilith-wsl-heavy.lock'),
               str(Path.home()/'work/claudehints/scripts/run-heavy'), '--mem', '12G', '--jobs', '1', '--',
               'env', 'TMPDIR='+str(Path.home()/'tmp'), sys.executable, '-B', __file__, '--admitted']
    raise SystemExit(subprocess.run(command).returncode)

assert sys.argv[1:] == ['--admitted']
sys.path.insert(0, str(Path('benchmarks/iqa_compete_2026-10-16').resolve()))
from compare_canonical import sha, compare
from reference_publication import register_outputs

def load(path):
    return json.loads(path.read_text())

def run(*command):
    subprocess.run(command, check=True)

def binding(path):
    return dict(path=str(path), sha256=sha(path))

assert not subprocess.check_output(['jj', 'diff', '--name-only'], text=True).strip()
control = load(CONTROL)
assert sha(R/'check_current_vmaf_studio4k_v1.py') == control['controller_sha256']
current_plan = R/'vmaf-fullcall-current-grid-run/plan.json'
assert sha(current_plan) == control['current_plan_sha256']
plan = load(current_plan)
source_paths = ['crates/vmaf', 'crates/zenmetrics-cli/src/metrics/vmaf.rs',
                'crates/zenmetrics-cli/src/metrics/classical.rs', 'crates/zenmetrics-cli/Cargo.toml',
                'crates/zenmetrics-math', 'Cargo.toml', 'Cargo.lock',
                'benchmarks/heaptrack/drivers/cpu_profile/Cargo.toml']
source_check = ['git', 'diff', '--name-only', plan['rust_source']+'..origin/master', '--', *source_paths]
assert not subprocess.check_output(source_check, text=True).strip()
control['publication_source_check'] = dict(command=source_check, differing_paths=[],
    origin_master=subprocess.check_output(['git', 'rev-parse', 'origin/master'], text=True).strip(),
    qualification='These metric, RGB ingress, math, manifest and dependency paths are unchanged from the EA worker source. Scheduling-table changes are separate from its explicitly selected metric calls.')
assert len(control['rows']) == 4
resource_log = R/'vmaf-studio4k-current-score-controls-v1.log'
resources = [line for line in resource_log.read_text().splitlines() if line.startswith('run-heavy: done ')]
assert len(resources) == 1 and 'rc=0 ' in resources[0]
control['publication_resources'] = dict(raw_result=binding(CONTROL), resource_log=binding(resource_log), receipt=resources[0])
for row in control['rows']:
    assert row['native_source'] == plan['rust_source'] and row['absolute_bound'] == 0.02
    comparisons = [compare(row['native_score'], score, row['absolute_bound'], 0) for score in row['retained_canonical_scores']]
    assert row['max_abs_delta'] == max(delta for delta, passed in comparisons)
    assert row['passed'] == all(passed is True for delta, passed in comparisons)

source_root = Path.home()/'tmp/portcheck/sources/libvmaf'
archive = source_root/'source.tar.gz'
archive_hash = sha(archive)
assert archive_hash == 'efe6ac72bb84ea796280e198f1d81454f92974e3fb4d57b021fcd9b0ff640615'
assert archive_hash in Path('Cargo.lock').read_text()
package = source_root/'vmaf-head-sys-0.2.0'
meson = package/'vendored/vmaf/libvmaf/meson.build'
vcs = package/'.cargo_vcs_info.json'
with tarfile.open(archive) as stream:
    for path in [meson, vcs]:
        member = 'vmaf-head-sys-0.2.0/'+str(path.relative_to(package))
        assert stream.extractfile(member).read() == path.read_bytes()
assert "version : '3.2.0'" in meson.read_text()
build = Path.home()/'tmp/portcheck/native/vmaf/build'
build_info = build/'meson-info/intro-projectinfo.json'
assert load(build_info)['version'] == '3.2.0'
header = build/'include/libvmaf/version.h'
assert '#define VMAF_API_VERSION_PATCH 0' in header.read_text()

               matching_lockfile_checksum=archive_hash, package_vcs=load(vcs),
               source_meson=binding(meson), source_version='3.2.0',
               build_project_info=binding(build_info), build_project=load(build_info), generated_header=binding(header),
               qualification='Measured effective oracle identity. Both archived source and linked build metadata identify libvmaf 3.2.0. Existing native descriptions saying 3.2.1 are retained pending documentation correction; they do not establish a different linked oracle version.')

                              ' codex-root-canonical publishing current-source VMAF controls\n')
